#include <iostream> //подключаем библиотеку ввода-вывода

int main() { //главная функция, с которой начинается выполнение программы
    int a = 4, b = 5; //объявляем две целочисленные переменные: a=4, b=5

    a = a + b; // складываем a и b, результат кладём в a. => a= 4 + 5 = 9
    b = a - b; // из нового а вычитаем b и кладём в b => b=4, a=9
    a = a - b; // из a вычитаем новое b (получаем исходное b (5) и кладем в a => a=5

}//конец функции main

#include <iostream>

int main() {
    int n = 5; //верхняя граница
    int f = 1, fact = 1;//две переменные f - итоговое произведение всех факториалов, fact - текущий факториал 

    for (int i = 1; i <= n; i++) //цикл от 1 до n включит. На каждом шаге i - очередное число
        f *= fact *= i; //обновляем текущий факториал и домотаем свежий факториал в общий результат (сразу два действия)

    std::cout << f; //печатаем итог
    return 0; //завершение программы
} //конец main
